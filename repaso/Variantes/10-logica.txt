Función peticion (genérica)

JavaScript

async function peticion(url, method, body) {

    Función reutilizable para cualquier petición HTTP. Recibe la URL, el método y opcionalmente un body.

JavaScript

    var opciones = {
        method: method,
        headers: { "Content-Type": "application/json" }
    };

    Configuramos las opciones de fetch. La cabecera dice que enviamos JSON.

JavaScript

    if (body && method !== "GET") {
        opciones.body = JSON.stringify(body);
    }

    Solo añadimos body si hay datos Y no es GET (GET no lleva body). JSON.stringify() convierte el objeto JavaScript a texto JSON.

JavaScript

    var response = await fetch(url, opciones);

    fetch() hace la petición HTTP. await espera a que el servidor responda. response es el objeto de respuesta.

JavaScript

    var data = await response.json();

    response.json() lee el cuerpo de la respuesta y lo parsea como JSON. También es asíncrono, por eso el await.

JavaScript

    if (!response.ok) {

    response.ok es true si el código HTTP está entre 200-299. Si es 400, 404, 500, etc., es false.

JavaScript

        throw new Error(data.error || "Error del servidor");

    throw new Error() lanza una excepción que será capturada por el catch de quien llamó a esta función. Usamos el mensaje de error del servidor si existe.

JavaScript

    }
    return data;
}

    Si todo fue bien, devolvemos los datos parseados.

Función validar

JavaScript

function validar(datos) {
    var errores = {};

    Objeto vacío donde acumulamos errores como { campo: "mensaje" }.

JavaScript

    if (!datos.codigo || datos.codigo.trim() === "") {
        errores.codigo = "El código es obligatorio";
    } else if (datos.codigo.length !== 9) {
        errores.codigo = "Debe tener exactamente 9 caracteres (tiene " + datos.codigo.length + ")";
    } else if (!/^[a-zA-Z0-9]{9}$/.test(datos.codigo)) {
        errores.codigo = "Solo letras y números";
    }

    Valida el código en 3 niveles:

        Que no esté vacío
        Que tenga exactamente 9 caracteres
        Que solo contenga letras y números (regex)

    La regex /^[a-zA-Z0-9]{9}$/:

        ^ → inicio del string
        [a-zA-Z0-9] → una letra o número
        {9} → exactamente 9 veces
        $ → fin del string
        .test() → devuelve true/false

JavaScript

    if (!datos.nombre || datos.nombre.trim() === "") {
        errores.nombre = "El nombre es obligatorio";
    } else if (datos.nombre.length > 100) {
        errores.nombre = "Máximo 100 caracteres";
    }

    Nombre: obligatorio y máximo 100 caracteres.

JavaScript

    if (!datos.talla || datos.talla === "") {
        errores.talla = "Selecciona una talla";
    } else if (TALLAS.indexOf(datos.talla.toUpperCase()) === -1) {
        errores.talla = "Talla no válida";
    }

    indexOf() busca el valor en el array. Devuelve -1 si no lo encuentra.

JavaScript

    if (datos.precio === "" || datos.precio === null) {
        errores.precio = "El precio es obligatorio";
    } else if (isNaN(parseFloat(datos.precio)) || parseFloat(datos.precio) <= 0) {
        errores.precio = "Debe ser un número mayor que 0";
    }

    isNaN() → "is Not a Number". Comprueba que el valor convertido a float sea un número válido y mayor que 0.

JavaScript

    if (!datos.email_creador || datos.email_creador.trim() === "") {
        errores.email_creador = "El email es obligatorio";
    } else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(datos.email_creador)) {
        errores.email_creador = "Formato de email no válido";
    }

    Regex del email /^[^\s@]+@[^\s@]+\.[^\s@]+$/:

        [^\s@]+ → uno o más caracteres que NO sean espacio ni @
        @ → la arroba literal
        [^\s@]+ → dominio
        \. → un punto literal
        [^\s@]+ → extensión (com, es, etc.)

JavaScript

    return errores;
}

    Devuelve el objeto. Si está vacío {} → todo correcto.

function leerFormulario() {
    return {
        id:            inputId.value || null,
        codigo:        document.getElementById("codigo").value.trim(),
        nombre:        document.getElementById("nombre").value.trim(),
        talla:         document.getElementById("talla").value,
        precio:        document.getElementById("precio").value,
        email_creador: document.getElementById("email_creador").value.trim()
    };
}

    Lee cada campo y devuelve un objeto. trim() quita espacios al inicio/final. inputId.value || null → si está vacío devuelve null (modo crear).

JavaScript

function limpiarErrores() {
    CAMPOS.forEach(function(campo) {
        var input = document.getElementById(campo);
        var span  = document.getElementById("error-" + campo);
        if (input) input.classList.remove("invalido");
        if (span)  span.textContent = "";
    });
}

    Recorremos todos los campos, quitamos la clase invalido y vaciamos los mensajes de error.
